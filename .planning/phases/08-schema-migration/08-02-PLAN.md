2. Add upgrade handler in getDb function:
```fsharp
let private getDb () : JS.Promise<obj> =
    let upgradeConfig =
        createObj [
            "upgrade" ==> fun (db: obj) (oldVersion: int) (newVersion: int) ->
                printfn "IndexedDB upgrade: %d → %d" oldVersion newVersion

                // v1 → v2: Clear existing queue (safest migration path)
                if oldVersion < 2 then
                    if db?objectStoreNames?contains(storeName) then
                        let tx = db?transaction([| storeName |], "readwrite")
                        let store = tx?objectStore(storeName)
                        store?clear() |> ignore
                        printfn "Cleared v1 queue during upgrade to v2"

                    // Object store already exists from v1, no schema change needed
                    // IndexedDB is schemaless for field names
                    ()
        ]
    openDB dbName dbVersion upgradeConfig
```

3. Update enqueue function to use new QueuedOperation fields:
```fsharp
let enqueue (operationType: OperationType) (userId: string) (workoutDate: string) : JS.Promise<QueueResult> =
    promise {
        try
            let! db = getDb()

            let operation = {
                id = None
                operationType = match operationType with
                                | OperationType.Create -> "CreateWorkout"
                                | OperationType.Delete -> "DeleteWorkout"
                recordId = None  // NEW: Will be used for update/delete in future
                userId = userId
                workoutDate = workoutDate
                recordType = "workout"  // NEW: Default to workout type
                textContent = None      // NEW: For future text records
                photoUrl = None         // NEW: For future photo records
                timestamp = System.DateTime.Now.ToUnixTimeMilliseconds()
                retryCount = 0
            }

            let! id = db?add(storeName, operation)
            return QueueResult.Success id
        with
        | ex -> return QueueResult.Error ex.Message
    }
```

Why clear queue: Current v1 queue operations reference old schema (onConflict). Safest to clear during upgrade. Users can re-toggle workouts if needed.

Alternative considered: Migrate each operation individually. Rejected due to complexity and low value for ~20 users.
