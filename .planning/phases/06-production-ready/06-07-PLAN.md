    ```typescript
    import { describe, it, expect } from 'vitest';
    import * as fs from 'fs';
    import * as path from 'path';

    const distPath = path.join(__dirname, '..', 'dist');
    const srcPath = path.join(__dirname, '..', 'src');
    const migrationsPath = path.join(__dirname, '..', 'supabase', 'migrations');

    describe('Phase 6: Production Ready', () => {

      describe('PWA Requirements (TECH-03)', () => {

        it('vite.config.ts includes VitePWA plugin', () => {
          const configPath = path.join(__dirname, '..', 'vite.config.ts');
          const config = fs.readFileSync(configPath, 'utf-8');
          expect(config).toContain('VitePWA');
          expect(config).toContain('registerType');
        });

        it('PWA icons exist in public folder', () => {
          const publicPath = path.join(__dirname, '..', 'public');
          expect(fs.existsSync(path.join(publicPath, 'pwa-192x192.png'))).toBe(true);
          expect(fs.existsSync(path.join(publicPath, 'pwa-512x512.png'))).toBe(true);
        });

        it('service worker registration module exists', () => {
          expect(fs.existsSync(path.join(srcPath, 'sw', 'Registration.fs'))).toBe(true);
        });

        it('manifest includes required fields for installability', () => {
          const configPath = path.join(__dirname, '..', 'vite.config.ts');
          const config = fs.readFileSync(configPath, 'utf-8');
          // Check manifest has required fields
          expect(config).toContain('name:');
          expect(config).toContain('short_name:');
          expect(config).toContain('icons:');
          expect(config).toContain('display:');
          expect(config).toContain('start_url:');
        });
      });

      describe('Offline Queue (TECH-02)', () => {

        it('idb library is installed', () => {
          const packageJson = JSON.parse(
            fs.readFileSync(path.join(__dirname, '..', 'package.json'), 'utf-8')
          );
          expect(packageJson.dependencies.idb).toBeDefined();
        });

        it('offline queue module exists', () => {
          expect(fs.existsSync(path.join(srcPath, 'offline', 'Queue.fs'))).toBe(true);
          expect(fs.existsSync(path.join(srcPath, 'offline', 'Types.fs'))).toBe(true);
        });

        it('sync module exists with Background Sync support', () => {
          const syncPath = path.join(srcPath, 'offline', 'Sync.fs');
          expect(fs.existsSync(syncPath)).toBe(true);
          const syncContent = fs.readFileSync(syncPath, 'utf-8');
          expect(syncContent).toContain('registerBackgroundSync');
          expect(syncContent).toContain('replayQueue');
        });

        it('network status module exists', () => {
          expect(fs.existsSync(path.join(srcPath, 'offline', 'NetworkStatus.fs'))).toBe(true);
        });

        it('offline indicator component exists', () => {
          expect(fs.existsSync(path.join(srcPath, 'Components', 'OfflineIndicator.fs'))).toBe(true);
        });
      });

      describe('Admin RBAC (ADMN-01, ADMN-02)', () => {

        it('admin RBAC migration exists', () => {
          const migrations = fs.readdirSync(migrationsPath);
          const adminMigration = migrations.find(m => m.includes('admin_rbac'));
          expect(adminMigration).toBeDefined();
        });

        it('admin migration includes user_roles table', () => {
          const migrations = fs.readdirSync(migrationsPath);
          const adminMigration = migrations.find(m => m.includes('admin_rbac'));
          const content = fs.readFileSync(
            path.join(migrationsPath, adminMigration!),
            'utf-8'
          );
          expect(content).toContain('CREATE TABLE');
          expect(content).toContain('user_roles');
          expect(content).toContain('is_admin');
        });

        it('admin migration includes DELETE policy for profiles', () => {
          const migrations = fs.readdirSync(migrationsPath);
          const adminMigration = migrations.find(m => m.includes('admin_rbac'));
          const content = fs.readFileSync(
            path.join(migrationsPath, adminMigration!),
            'utf-8'
          );
          expect(content).toContain('CREATE POLICY');
          expect(content).toContain('DELETE');
          expect(content).toContain('is_admin');
        });

        it('admin API module exists', () => {
          expect(fs.existsSync(path.join(srcPath, 'Supabase', 'Admin.fs'))).toBe(true);
        });

        it('admin page exists with access control', () => {
          const adminPagePath = path.join(srcPath, 'Pages', 'AdminPage.fs');
          expect(fs.existsSync(adminPagePath)).toBe(true);
          const content = fs.readFileSync(adminPagePath, 'utf-8');
          expect(content).toContain('isAdmin');
          expect(content).toContain('NotAdmin');
        });

        it('member list component exists', () => {
          expect(fs.existsSync(path.join(srcPath, 'admin', 'MemberList.fs'))).toBe(true);
        });

        it('delete confirmation component exists', () => {
          const actionsPath = path.join(srcPath, 'admin', 'MemberActions.fs');
          expect(fs.existsSync(actionsPath)).toBe(true);
          const content = fs.readFileSync(actionsPath, 'utf-8');
          expect(content).toContain('DeleteConfirmModal');
        });
      });

      describe('Bundle Size Optimization', () => {

        it('rollup-plugin-visualizer is installed', () => {
          const packageJson = JSON.parse(
            fs.readFileSync(path.join(__dirname, '..', 'package.json'), 'utf-8')
          );
          expect(packageJson.devDependencies['rollup-plugin-visualizer']).toBeDefined();
        });

        it('vite.config.ts includes manual chunks', () => {
          const configPath = path.join(__dirname, '..', 'vite.config.ts');
          const config = fs.readFileSync(configPath, 'utf-8');
          expect(config).toContain('manualChunks');
        });

        it('vite.config.ts sets chunk size warning limit', () => {
          const configPath = path.join(__dirname, '..', 'vite.config.ts');
          const config = fs.readFileSync(configPath, 'utf-8');
          expect(config).toContain('chunkSizeWarningLimit');
        });
      });

      describe('Security', () => {

        it('all migrations enable RLS', () => {
          const migrations = fs.readdirSync(migrationsPath).filter(f => f.endsWith('.sql'));
          for (const migration of migrations) {
            const content = fs.readFileSync(path.join(migrationsPath, migration), 'utf-8');
            // If migration creates a table, it should enable RLS
            if (content.includes('CREATE TABLE') && content.includes('public.')) {
              expect(content).toContain('ENABLE ROW LEVEL SECURITY');
            }
          }
        });

        it('storage bucket migration has RLS policies', () => {
          const migrations = fs.readdirSync(migrationsPath);
          const storageMigration = migrations.find(m => m.includes('storage'));
          if (storageMigration) {
            const content = fs.readFileSync(
              path.join(migrationsPath, storageMigration),
              'utf-8'
            );
            expect(content).toContain('CREATE POLICY');
            expect(content).toContain('storage.objects');
          }
        });

        it('no service role key in source code', () => {
          const checkFile = (filePath: string) => {
            const content = fs.readFileSync(filePath, 'utf-8');
            expect(content).not.toContain('service_role');
            expect(content).not.toContain('SERVICE_ROLE');
          };

          // Check client files
          const clientPath = path.join(srcPath, 'Supabase', 'Client.fs');
          if (fs.existsSync(clientPath)) {
            checkFile(clientPath);
          }
        });
      });

    });
    ```
