Test RLS policies by simulating user contexts:

```bash
docker exec -i supabase_db_rollbook psql -U postgres -d postgres << 'EOF'
-- Create two test users

VALUES
  ('aaaaaaaa-aaaa-aaaa-aaaa-aaaaaaaaaaaa', 'storage-test-a@test.com', '00000000-0000-0000-0000-000000000000', 'authenticated', 'authenticated', now(), now(), '', now()),
  ('bbbbbbbb-bbbb-bbbb-bbbb-bbbbbbbbbbbb', 'storage-test-b@test.com', '00000000-0000-0000-0000-000000000000', 'authenticated', 'authenticated', now(), now(), '', now())
ON CONFLICT (id) DO NOTHING;

-- Test 3a: Insert mock storage object as User A (via service role, simulating upload result)
-- Note: Actual uploads go through Storage API, but we test RLS SELECT/DELETE logic
INSERT INTO storage.objects (bucket_id, name, owner, created_at, updated_at)
VALUES ('workout-photos', 'aaaaaaaa-aaaa-aaaa-aaaa-aaaaaaaaaaaa/2026-02-10.jpg', 'aaaaaaaa-aaaa-aaaa-aaaa-aaaaaaaaaaaa', now(), now())
ON CONFLICT DO NOTHING;

-- Test 3b: User A can SELECT own file
SET LOCAL ROLE authenticated;
SET LOCAL request.jwt.claims = '{"sub": "aaaaaaaa-aaaa-aaaa-aaaa-aaaaaaaaaaaa", "role": "authenticated"}';

SELECT count(*) as user_a_own_files
FROM storage.objects
WHERE bucket_id = 'workout-photos'
  AND name LIKE 'aaaaaaaa-aaaa-aaaa-aaaa-aaaaaaaaaaaa/%';
-- Expected: 1

RESET ROLE;

-- Test 3c: User B cannot SELECT User A's file
SET LOCAL ROLE authenticated;
SET LOCAL request.jwt.claims = '{"sub": "bbbbbbbb-bbbb-bbbb-bbbb-bbbbbbbbbbbb", "role": "authenticated"}';

SELECT count(*) as user_b_sees_user_a_files
FROM storage.objects
WHERE bucket_id = 'workout-photos'
  AND name LIKE 'aaaaaaaa-aaaa-aaaa-aaaa-aaaaaaaaaaaa/%';
-- Expected: 0 (RLS blocks access)

RESET ROLE;

-- Test 3d: User B cannot DELETE User A's file
SET LOCAL ROLE authenticated;
SET LOCAL request.jwt.claims = '{"sub": "bbbbbbbb-bbbb-bbbb-bbbb-bbbbbbbbbbbb", "role": "authenticated"}';

DELETE FROM storage.objects
WHERE bucket_id = 'workout-photos'
  AND name = 'aaaaaaaa-aaaa-aaaa-aaaa-aaaaaaaaaaaa/2026-02-10.jpg';
-- Expected: DELETE 0 (RLS blocks)

RESET ROLE;

-- Test 3e: User A CAN delete own file
SET LOCAL ROLE authenticated;
SET LOCAL request.jwt.claims = '{"sub": "aaaaaaaa-aaaa-aaaa-aaaa-aaaaaaaaaaaa", "role": "authenticated"}';

DELETE FROM storage.objects
WHERE bucket_id = 'workout-photos'
  AND name = 'aaaaaaaa-aaaa-aaaa-aaaa-aaaaaaaaaaaa/2026-02-10.jpg';
-- Expected: DELETE 1

RESET ROLE;

-- Cleanup test users
DELETE FROM auth.users WHERE email LIKE 'storage-test-%@test.com';

SELECT 'All RLS tests passed' as result;
EOF
```
