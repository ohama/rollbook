**TeamMemberRow component:**
Renders a single team member with their stats:
- Props: stats (TeamMemberStats), isCurrentUser (bool)
- Left side: Emoji indicator (workout_count > 0) + display_name (+ "(나)" if current user)
- Right side: workout_count number with "회" suffix
- Conditional styling: bg-indigo-50 border for current user, bg-white for others

```fsharp
[<ReactComponent>]
let TeamMemberRow (stats: TeamMemberStats) (isCurrentUser: bool) =
    Html.div [
        prop.className (
            "flex items-center justify-between p-4 " +
            if isCurrentUser then "bg-indigo-50 border border-indigo-200 rounded-lg"
            else "bg-white"
        )
        prop.children [
            // Left: emoji + name
            Html.div [
                prop.className "flex items-center gap-3"
                prop.children [
                    Html.span [
                        prop.className "text-2xl"
                        prop.text (if stats.workout_count > 0 then "..." else "...")
                    ]
                    Html.span [
                        prop.className (if isCurrentUser then "font-medium text-indigo-800" else "font-medium text-gray-800")
                        prop.text (if isCurrentUser then stats.display_name + " (나)" else stats.display_name)
                    ]
                ]
            ]
            // Right: count
            Html.div [
                prop.className "text-right"
                prop.children [
                    Html.span [
                        prop.className "text-2xl font-bold text-indigo-600"
                        prop.text (string stats.workout_count)
                    ]
                    Html.span [
                        prop.className "text-sm text-gray-500 ml-1"
                        prop.text "회"
                    ]
                ]
            ]
        ]
    ]
```

Main page with month navigation and roster:
- Props: currentUserId (string) - passed from Dashboard for highlighting
- State: currentYear, currentMonth, teamStats, loading, error (same pattern as ProgressView)
- Month navigation: goToNextMonth, goToPrevMonth (same as ProgressView)
- useEffect: fetch team stats when month changes

```fsharp
[<ReactComponent>]
let TeamViewPage (currentUserId: string) =
    // Month navigation state
    let (currentYear, setCurrentYear) = React.useState(System.DateTime.Now.Year)
    let (currentMonth, setCurrentMonth) = React.useState(System.DateTime.Now.Month)

    // Data state
    let (teamStats, setTeamStats) = React.useState<TeamMemberStats array>([||])
    let (loading, setLoading) = React.useState(true)
    let (error, setError) = React.useState<string option>(None)

    // Month navigation (same as ProgressView)
    let goToNextMonth () = ...
    let goToPrevMonth () = ...

    // Fetch team stats on month change
    React.useEffect((fun () ->
        promise {
            try
                setLoading true
                setError None
                let! stats = getTeamMonthlyStats currentYear currentMonth
                setTeamStats stats
                setLoading false
            with ex ->
                setError (Some "팀 통계를 불러올 수 없습니다")
                setLoading false
        } |> Promise.start
    ), [| box currentYear; box currentMonth |])
```
