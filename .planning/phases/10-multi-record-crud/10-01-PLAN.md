```fsharp
let createPhotoRecord (userId: string) (date: string) (photoUrl: string) (textContent: string option) : JS.Promise<WorkoutResponse> =
```

3. **updateWorkoutById** — Updates a specific record by id (text_content only):

```fsharp
/// Update a specific workout record by id (for editing text content)
let updateWorkoutById (recordId: int) (textContent: string) : JS.Promise<WorkoutResponse> =
    promise {
        let nowIso : string = emitJsExpr () "new Date().toISOString()"
        let updates = createObj [
            "text_content" ==> textContent
            "updated_at" ==> nowIso
        ]
        let query =
            supabase
                ?from("workouts")
                ?update(updates)
                ?eq("id", recordId)
                ?is("deleted_at", null)
                ?select()
        let! result = query
        return unbox<WorkoutResponse> result
    }
```

Follow the exact same patterns as existing functions (upsertWorkout, deleteWorkoutById). Use `?` operator for Supabase dynamic interop. Include soft delete filter (`?is("deleted_at", null)`) on update.
